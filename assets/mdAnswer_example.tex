% =========================================================
\newcommand{\SafeIncludeGraphics}[2][]{%
  \IfFileExists{\detokenize{#2}}{%
    \includegraphics[#1]{\detokenize{#2}}%
  }{%
    \begin{tcolorbox}[
      colback=white,
      colframe=red!60!black,
      boxrule=0.4pt,
      sharp corners,
      left=1mm,right=1mm,top=1mm,bottom=1mm
    ]
      \textbf{[画像が見つかりません]}
    \end{tcolorbox}%
  }%
}
